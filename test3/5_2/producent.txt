import java.io.*;
import java.net.*;
import java.util.*;

public class SerwerBiblioteki {
    private static List<Ksiazka> listaKsiazek = new ArrayList<>();
    private static int generatorId = 1;

    public static void main(String[] args) throws IOException {
        ServerSocket serwer = new ServerSocket(5000);
        System.out.println("Serwer Biblioteki uruchomiony na porcie 5000...");

        while (true) {
            Socket klient = serwer.accept();
            // Obsługa każdego klienta w nowym wątku
            new Thread(() -> obsluzZgloszenie(klient)).start();
        }
    }

    private static void obsluzZgloszenie(Socket s) {
        try (BufferedReader in = new BufferedReader(new InputStreamReader(s.getInputStream()));
             PrintWriter out = new PrintWriter(s.getOutputStream(), true)) {

            String zadanie;
            while ((zadanie = in.readLine()) != null) {
                String[] p = zadanie.split(";"); // np. WYPOZYCZ;Jan;1
                String komenda = p[0];
                String user = p[1];

                synchronized (listaKsiazek) {
                    // Tutaj szukamy obiektu Ksiazka w liście i wykonujemy operacje
                    // Przykład dla wypożyczenia:
                    if (komenda.equals("WYPOZYCZ")) {
                        int id = Integer.parseInt(p[2]);
                        Ksiazka k = znajdz(id);
                        if (k != null && !k.czyZablokowana) {
                            k.aktualniCzytelnicy.add(user);
                            out.println("Sukces;Tresc:" + k.tresc);
                        } else {
                            out.println("Blad;Ksiazka niedostepna");
                        }
                    } 
                    // ... reszta komend (DODAJ, BLOKUJ, USUN, RAPORT)
                }
            }
        } catch (Exception e) { System.out.println("Klient rozlaczony"); }
    }

    private static Ksiazka znajdz(int id) {
        for (Ksiazka k : listaKsiazek) if (k.nrBiblioteczny == id) return k;
        return null;
    }
}
