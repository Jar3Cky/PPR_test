import java.util.ArrayList;
import java.util.List;

public class Ksiazka {
    public int nrBiblioteczny;
    public String autor;
    public String tytul;
    public String dataWydania;
    public String tresc;
    public String wlasciciel; // login pisarza, który ją dodał

    public boolean czyZablokowana = false; // dla pisarza (do niszczenia/edycji)
    public List<String> aktualniCzytelnicy = new ArrayList<>(); // lista loginów osób czytających

    public Ksiazka(int nr, String autor, String tytul, String data, String tresc, String wlasciciel) {
        this.nrBiblioteczny = nr;
        this.autor = autor;
        this.tytul = tytul;
        this.dataWydania = data;
        this.tresc = tresc;
        this.wlasciciel = wlasciciel;
    }

    @Override
    public String toString() {
        return String.format("ID: %d | %s - %s | Blokada: %b | Czytelnicy: %s", 
                nrBiblioteczny, autor, tytul, czyZablokowana, aktualniCzytelnicy);
    }
}
