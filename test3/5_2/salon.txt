import java.io.*;
import java.net.*;
import java.util.Scanner;

public class KlientBiblioteki {
    public static void main(String[] args) throws IOException {
        Socket s = new Socket("localhost", 5000);
        PrintWriter out = new PrintWriter(s.getOutputStream(), true);
        BufferedReader in = new BufferedReader(new InputStreamReader(s.getInputStream()));
        Scanner skaner = new Scanner(System.in);

        System.out.println("Witaj w Bibliotece Rozproszonej!");
        System.out.print("Podaj swoj nick: ");
        String nick = skaner.nextLine();

        // Prosta pętla do wysyłania komend
        while (true) {
            System.out.println("Wpisz komende (np. WYPOZYCZ;1 lub RAPORT):");
            String cmd = skaner.nextLine();
            out.println(cmd.toUpperCase() + ";" + nick); // wysyłamy do serwera
            System.out.println("Serwer odpowiedzial: " + in.readLine());
        }
    }
}
