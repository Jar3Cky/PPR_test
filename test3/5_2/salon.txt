import java.io.*;
import java.net.*;
import java.util.*;

public class SalonSerwer {
    private static final int PORT = 5000;
    private static final int POJEMNOSC = 5;
    private static final List<String> magazyn = new ArrayList<>();

    public static void main(String[] args) throws IOException {
        ServerSocket serverSocket = new ServerSocket(PORT);
        System.out.println("[*] Salon otwarty na porcie " + PORT);

        while (true) {
            Socket clientSocket = serverSocket.accept();
            new Thread(new ClientHandler(clientSocket)).start();
        }
    }

    static class ClientHandler implements Runnable {
        private Socket socket;

        public ClientHandler(Socket socket) { this.socket = socket; }

        @Override
        public void run() {
            try (BufferedReader in = new BufferedReader(new InputStreamReader(socket.getInputStream()));
                 PrintWriter out = new PrintWriter(socket.getOutputStream(), true)) {

                String request = in.readLine();
                if (request == null) return;

                String[] parts = request.split("\\|");
                String typ = parts[0]; // PRODUCENT lub KONSUMENT
                String dane = parts[1]; // Marka lub "ANY"

                synchronized (magazyn) {
                    if (typ.equals("PRODUCENT")) {
                        if (magazyn.size() < POJEMNOSC) {
                            magazyn.add(dane); // dane to np. "Audi (VIN: 1234)"
                            System.out.println("[LOG] Przyjęto: " + dane);
                            out.println("OK|Samochód dostarczony");
                        } else {
                            out.println("WAIT|Brak miejsca w salonie");
                        }
                    } else if (typ.equals("KONSUMENT")) {
                        Optional<String> znaleziony = (dane.equals("ANY")) 
                            ? magazyn.stream().findFirst() 
                            : magazyn.stream().filter(s -> s.startsWith(dane)).findFirst();

                        if (znaleziony.isPresent()) {
                            magazyn.remove(znaleziony.get());
                            System.out.println("[LOG] Sprzedano: " + znaleziony.get());
                            out.println("OK|Kupiono: " + znaleziony.get());
                        } else {
                            out.println("RETRY|Brak wybranego modelu");
                        }
                    }
                }
            } catch (IOException e) {
                e.printStackTrace();
            }
        }
    }
}
