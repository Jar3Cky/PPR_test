import java.util.LinkedList;
import java.util.Queue;

public class Salon {
    private final int capacity;
    private final Queue<Car> inventory = new LinkedList<>();

    public Salon(int capacity) {
        this.capacity = capacity;
    }

    // Metoda dla Producenta
    public synchronized void put(Car car, String producerName) throws InterruptedException {
        while (inventory.size() >= capacity) {
            System.out.println("[SALON] Pełny! " + producerName + " wstrzymuje produkcję.");
            wait(); // Czekaj na wolne miejsce
        }
        inventory.add(car);
        System.out.println("[PRODUKCJA] " + producerName + " wstawił: " + car);
        notifyAll(); // Powiadom wszystkich (klientów), że pojawiło się auto
    }

    // Metoda dla Konsumenta (Punkt 3 - czekanie na konkretną markę)
    public synchronized Car takeSpecific(String consumerName, String preferredBrand) throws InterruptedException {
        while (true) {
            Car foundCar = null;
            // Szukamy wybranej marki w inwentarzu
            for (Car car : inventory) {
                if (car.getBrand().equalsIgnoreCase(preferredBrand)) {
                    foundCar = car;
                    break;
                }
            }

            if (foundCar != null) {
                inventory.remove(foundCar);
                System.out.println("[ZAKUP] " + consumerName + " nabył: " + foundCar);
                notifyAll(); // Powiadom producentów, że zwolniło się miejsce
                return foundCar;
            }

            // Jeśli nie ma marki, klient nie wychodzi, tylko czeka (Zadanie 3)
            System.out.println("[KLIENT] " + consumerName + " czeka na model: " + preferredBrand);
            wait(); 
        }
    }
}
