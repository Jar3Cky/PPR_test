import java.io.*;
import java.net.*;

public class RemoteConsumer {
    public static void main(String[] args) throws Exception {
        String serverIP = "ADRES_IP_SERWERA";
        String preferredBrand = "BMW";

        while (true) {
            try (Socket socket = new Socket(serverIP, 12345);
                 ObjectOutputStream out = new ObjectOutputStream(socket.getOutputStream());
                 ObjectInputStream in = new ObjectInputStream(socket.getInputStream())) {

                out.writeObject("BUY");
                out.writeObject(preferredBrand);

                Object response = in.readObject();
                if (response instanceof Car) {
                    System.out.println("Kupiłem: " + response);
                } else {
                    System.out.println("Brak " + preferredBrand + ", spróbuję za chwilę...");
                }
            }
            Thread.sleep(4000);
        }
    }
}
